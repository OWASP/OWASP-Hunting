# Pause-based local socket fixture: one pair, 250 ms pause
import socket, time
with socket.create_connection(('127.0.0.1', 8080), timeout=3) as desync_socket:
    desync_socket.sendall(b'POST /desync-lab/pause HTTP/1.1\r\nHost: localhost\r\nContent-Length: 4\r\nConnection: keep-alive\r\n\r\n')
    time.sleep(0.25)
    desync_socket.sendall(b'PINGGET /desync-lab/pause-marker HTTP/1.1\r\nHost: localhost\r\nConnection: close\r\n\r\n')
    print(desync_socket.recv(4096))
