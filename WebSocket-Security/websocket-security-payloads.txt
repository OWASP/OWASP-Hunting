# WebSocket Security Payloads

const wsOriginProbe = new WebSocket('wss://app.example.test/ws/security-lab', ['security-lab.v1']);
wsOriginProbe.addEventListener('open', () => wsOriginProbe.send(JSON.stringify({type:'whoami',requestId:'ws-origin-control'})));
wsOriginProbe.addEventListener('message', event => console.log('ws-origin-result', event.data));

# Lifecycle sequences
# Same-origin browser lab: keep the socket open across logout, then send a protected read
fetch('https://app.example.test/api/session-lab/logout',{method:'POST',credentials:'include'}).then(() => wsOriginProbe.send(JSON.stringify({type:'history',channel:'ws-room-a',limit:1,requestId:'ws-after-logout'})));
