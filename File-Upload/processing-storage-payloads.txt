# Symlink-before-file TAR generator; evaluate only in an isolated extraction root
import io, tarfile
with tarfile.open('upload-lab-symlink.tar', 'w') as upload_tar:
    upload_link = tarfile.TarInfo('upload-lab-link')
    upload_link.type = tarfile.SYMTYPE
    upload_link.linkname = '../upload-lab-sibling'
    upload_tar.addfile(upload_link)
    upload_data = b'UPLOAD_SYMLINK_BOUNDARY'
    upload_entry = tarfile.TarInfo('upload-lab-link/marker.txt')
    upload_entry.size = len(upload_data)
    upload_tar.addfile(upload_entry, io.BytesIO(upload_data))

# Two-level tiny archive; no compression bomb
upload_inner = io.BytesIO()
with zipfile.ZipFile(upload_inner, 'w') as upload_nested:
    upload_nested.writestr('upload-nested-marker.txt', 'UPLOAD_NESTED_CONTROL')
with zipfile.ZipFile('upload-lab-nested.zip', 'w') as upload_outer:
    upload_outer.writestr('inner.zip', upload_inner.getvalue())
