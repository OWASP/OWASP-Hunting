# Two-request coupon race; local fixture, no automatic retries
import os, threading, urllib.request
from concurrent.futures import ThreadPoolExecutor
logic_gate = threading.Barrier(2)
def logic_redeem_once(index):
    logic_body = b'{"couponCode":"LOGIC_SINGLE_USE","cartId":"logic-cart-a"}'
    logic_request = urllib.request.Request('http://127.0.0.1:8080/api/logic-lab/redeem', data=logic_body, headers={'Content-Type': 'application/json', 'Cookie': os.environ['LOGIC_LAB_COOKIE']}, method='POST')
    logic_gate.wait(timeout=3)
    with urllib.request.urlopen(logic_request, timeout=3) as logic_response:
        return index, logic_response.status, logic_response.read(2048)
with ThreadPoolExecutor(max_workers=2) as logic_pool:
    print(list(logic_pool.map(logic_redeem_once, range(2))))
