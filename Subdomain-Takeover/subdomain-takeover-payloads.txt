# Subdomain Takeover Payloads
# DNS checks, provider fingerprints and proof-of-control markers. Verify each fingerprint against the provider's current behaviour; they change.

# DNS reconnaissance commands
dig +short CNAME sub.example.test
dig +short NS sub.example.test
dig +short MX sub.example.test
dig +short A sub.example.test
dig sub.example.test ANY +noall +answer
host -t CNAME sub.example.test
nslookup -type=CNAME sub.example.test
dig +trace sub.example.test
dig @8.8.8.8 sub.example.test CNAME +short
dig +short CNAME $(dig +short CNAME sub.example.test)

# Subdomain collection
subfinder -d example.test -all -silent
amass enum -passive -d example.test
curl -s "https://crt.sh/?q=%25.example.test&output=json"
assetfinder --subs-only example.test
dnsx -l subdomains.txt -cname -resp -silent

# Takeover candidate triage
subjack -w subdomains.txt -t 100 -timeout 30 -ssl -c fingerprints.json -v
nuclei -l live-hosts.txt -t http/takeovers/
dnsReaper scan --filename subdomains.txt
httpx -l subdomains.txt -status-code -title -cname -silent
